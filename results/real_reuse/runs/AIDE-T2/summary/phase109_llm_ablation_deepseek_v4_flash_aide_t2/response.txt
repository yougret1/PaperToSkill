```python
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder, FunctionTransformer

# ---------- Data Loading ----------
train = pd.read_csv('train.csv')
valid = pd.read_csv('validation_features.csv')

# Separate features and target
target = 'Transported'
X_train = train.drop(columns=[target, 'PassengerId', 'Name'])
y_train = train[target].astype(int)  # Ensure integer 0/1
X_valid = valid.drop(columns=['PassengerId', 'Name'])

# ---------- Feature Engineering ----------

# Cabin: extract Deck, Num, Side (if present)
def parse_cabin(df):
    df = df.copy()
    if 'Cabin' in df.columns:
        parts = df['Cabin'].str.split('/', expand=True)
        df['Deck'] = parts[0].fillna('Unknown')
        df['CabinNum'] = pd.to_numeric(parts[1], errors='coerce')
        df['Side'] = parts[2].fillna('Unknown')
        df.drop(columns=['Cabin'], inplace=True)
    else:
        df['Deck'] = 'Unknown'
        df['CabinNum'] = np.nan
        df['Side'] = 'Unknown'
    return df

X_train = parse_cabin(X_train)
X_valid = parse_cabin(X_valid)

# Identify numeric and categorical columns
num_cols = ['Age', 'RoomService', 'FoodCourt', 'ShoppingMall', 'Spa', 'VRDeck', 'CabinNum']
cat_cols = ['HomePlanet', 'CryoSleep', 'Destination', 'VIP', 'Deck', 'Side']

# Filter only columns that actually exist in data
num_cols = [c for c in num_cols if c in X_train.columns]
cat_cols = [c for c in cat_cols if c in X_train.columns]

# ---------- Preprocessing Pipelines ----------
numeric_transformer = Pipeline([
    ('imputer', SimpleImputer(strategy='median')),
    ('scaler', StandardScaler())
])

categorical_transformer = Pipeline([
    ('imputer', SimpleImputer(strategy='constant', fill_value='missing')),
    ('onehot', OneHotEncoder(handle_unknown='ignore'))
])

preprocessor = ColumnTransformer([
    ('num', numeric_transformer, num_cols),
    ('cat', categorical_transformer, cat_cols)
])

# ---------- Model ----------
model = Pipeline([
    ('preproc', preprocessor),
    ('classifier', RandomForestClassifier(
        n_estimators=300,
        max_depth=12,
        min_samples_leaf=4,
        random_state=42,
        n_jobs=-1
    ))
])

# ---------- Training ----------
model.fit(X_train, y_train)

# ---------- Predict on Validation ----------
preds = model.predict(X_valid)  # Returns 0/1 int
preds_bool = preds.astype(bool)  # Convert to True/False

# ---------- Submission ----------
submission = pd.DataFrame({
    'PassengerId': valid['PassengerId'],
    'Transported': preds_bool
})
submission.to_csv('submission.csv', index=False)
```
