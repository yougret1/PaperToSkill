    budget = case["payload"]["budget"]

    n = len(segments)
    names = [s["name"] for s in segments]
    importance = [s["importance"] for s in segments]
    minimum = [s["minimum"] for s in segments]
    capacity = [s["capacity"] for s in segments]

    total_min = sum(minimum)
    if budget < total_min:
        return {"error": "budget insufficient to meet minimums"}

    # Start with minimums, then greedily assign remaining budget
    # to the segment with highest importance (up to its capacity)
    allocation = list(minimum)
    remaining = budget - total_min

    # Sort segment indices by importance descending
    order = sorted(range(n), key=lambda i: importance[i], reverse=True)

    for i in order:
        if remaining <= 0:
            break
        can_add = capacity[i] - allocation[i]
        add = min(can_add, remaining)
        allocation[i] += add
        remaining -= add

    return {
        "allocations": {names[i]: allocation[i] for i in range(n)},
        "total_allocated": sum(allocation),
        "total_importance": sum(importance[i] * allocation[i] for i in range(n)),
    }
