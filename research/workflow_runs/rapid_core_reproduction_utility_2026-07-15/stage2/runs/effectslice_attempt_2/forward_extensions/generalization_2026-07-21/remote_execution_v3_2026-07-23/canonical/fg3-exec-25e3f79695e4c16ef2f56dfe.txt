```python
def solve(case):
    # Validate required fields
    for field in ("nodes", "core_distance", "distances"):
        if field not in case:
            raise ValueError(f"Missing required field: '{field}'")

    nodes = list(case["nodes"])
    core_distance = dict(case["core_distance"])
    distances = case["distances"]
    n = len(nodes)

    if n == 0:
        raise ValueError("nodes list is empty")

    node_set = set(nodes)

    for node in nodes:
        if node not in core_distance:
            raise ValueError(f"Node '{node}' missing from core_distance")

    # Build mutual-reachability edges
    edges = []
    for entry in distances:
        if len(entry) != 3:
            raise ValueError(f"Distance entry must have 3 elements, got {len(entry)}")
        u, v, d = entry[0], entry[1], entry[2]
        if u not in node_set:
            raise ValueError(f"Distance references unknown node '{u}'")
        if v not in node_set:
            raise ValueError(f"Distance references unknown node '{v}'")
        # Canonicalize undirected endpoints
        eu, ev = (u, v) if u <= v else (v, u)
        mr = max(core_distance[eu], core_distance[ev], d)
        edges.append((mr, eu, ev))

    # Sort by reachability weight, then canonicalized endpoint IDs
    edges.sort(key=lambda e: (e[0], e[1], e[2]))

    # Union-Find with union-by-rank and path-halving
    parent = {node: node for node in nodes}
    rank = {node: 0 for node in nodes}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]  # path halving
            x = parent[x]
        return x

    def union(x, y):
        rx, ry = find(x), find(y)
        if rx == ry:
            return False
        if rank[rx] < rank[ry]:
            rx, ry = ry, rx
        parent[ry] = rx
        if rank[rx] == rank[ry]:
            rank[rx] += 1
        return True

    # Kruskal's MST
    mst_edges = []
    total_weight = 0.0

    for mr, eu, ev in edges:
        if union(eu, ev):
            mst_edges.append({"u": eu, "v": ev, "weight": mr})
            total_weight += mr
            if len(mst_edges) == n - 1:
                break

    if len(mst_edges) < n - 1:
        raise ValueError("Graph is disconnected; spanning tree is not possible")

    return {
        "mst_edges": mst_edges,
        "total_weight": total_weight,
    }
```

Key design decisions:

- Mutual reachability: `max(core[u], core[v], dist(u,v))` per the HDBSCAN definition.
- Canonicalization: endpoints are sorted lexicographically (`u <= v`) before any comparison or storage, making edge identity deterministic regardless of input order.
- Tie-breaking: edges are sorted by `(weight, u, v)` — fully deterministic since endpoints are canonicalized strings.
- Union-Find: union-by-rank with path-halving; rank determines merge direction, so ties in rank always promote the left root, keeping behavior stable.
- Disconnected graphs raise `ValueError` rather than returning a partial result.
- Input is never mutated — `list()` and `dict()` copies are made upfront.
