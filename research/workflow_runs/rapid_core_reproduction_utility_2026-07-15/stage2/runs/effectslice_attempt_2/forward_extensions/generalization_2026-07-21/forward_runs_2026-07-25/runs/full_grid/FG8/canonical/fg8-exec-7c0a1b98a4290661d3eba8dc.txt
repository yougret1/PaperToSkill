  "implementation": "def solve(case):\n    segments = case['segments']\n    budget = case['budget']\n    n = len(segments)\n    \n    # extract fields\n    names = [seg['name'] for seg in segments]\n    importances = [seg['importance'] for seg in segments]\n    minimums = [seg['minimum'] for seg in segments]\n    capacities = [seg['capacity'] for seg in segments]\n    \n    total_min = sum(minimums)\n    total_cap = sum(capacities)\n    \n    if budget < total_min or budget > total_cap:\n        raise ValueError('Budget out of feasible range')\n    \n    # initial allocation at minimum\n    alloc = list(minimums)\n    remaining = budget - total_min\n    \n    # current positions = current allocation + 1 (since we give next token)\n    # importance / (current_pos) where current_pos = alloc[i] + 1\n    # We'll update after each token\n    \n    while remaining > 0:\n        best_idx = -1\n        best_score = -1.0\n        for i in range(n):\n            if alloc[i] < capacities[i]:\n                # score = importance / (next position)\n                # next position = alloc[i] + 1\n                score = importances[i] / (alloc[i] + 1.0)\n                # Use tuple comparison with index as tiebreaker\n                if score > best_score or (score == best_score and (best_idx == -1 or i < best_idx)):\n                    best_score = score\n                    best_idx = i\n        if best_idx == -1:\n            # should not happen because total_cap >= budget\n            raise ValueError('Cannot allocate remaining budget')\n        alloc[best_idx] += 1\n        remaining -= 1\n    \n    # build output\n    by_name = {names[i]: alloc[i] for i in range(n)}\n    return {\n        'allocations': alloc,\n        'by_name': by_name,\n        'total': sum(alloc)\n    }"
