```python
def solve(case):
    # Validate required top-level fields
    if not isinstance(case, dict):
        raise ValueError("case must be a dict")
    payload = case.get("payload")
    if payload is None:
        raise ValueError("Missing 'payload' in case")
    if not isinstance(payload, dict):
        raise ValueError("'payload' must be a dict")

    nodes_data = payload.get("nodes")
    required_symbols = payload.get("required_symbols")

    if nodes_data is None:
        raise ValueError("Missing 'nodes' in payload")
    if required_symbols is None:
        raise ValueError("Missing 'required_symbols' in payload")
    if not isinstance(nodes_data, list):
        raise ValueError("'nodes' must be a list")
    if not isinstance(required_symbols, list):
        raise ValueError("'required_symbols' must be a list")

    required_fields = ("id", "parent", "symbol", "depth", "removable", "utility")

    # Build node registry (copy each node to avoid mutating input)
    nodes = {}
    for raw in nodes_data:
        for field in required_fields:
            if field not in raw:
                raise ValueError(f"Node missing required field '{field}': {raw}")
        nid = raw["id"]
        if nid in nodes:
            raise ValueError(f"Duplicate node id '{nid}'")
        nodes[nid] = {k: raw[k] for k in required_fields}

    # Build parent→children map
    children = {nid: [] for nid in nodes}
    for nid, n in nodes.items():
        parent = n["parent"]
        if parent is not None:
            if parent not in nodes:
                raise ValueError(f"Node '{nid}' references unknown parent '{parent}'")
            children[parent].append(nid)

    # Validate exactly one root
    roots = [nid for nid, n in nodes.items() if n["parent"] is None]
    if len(roots) != 1:
        raise ValueError(f"Expected exactly one root, found {len(roots)}: {roots}")

    required_symbols_set = set(required_symbols)

    # Validate all required symbols are present in the initial tree
    initial_symbols = {n["symbol"] for n in nodes.values()}
    missing = required_symbols_set - initial_symbols
    if missing:
        raise ValueError(f"Required symbols not present in tree: {missing}")

    # Working set of retained node IDs
    retained = set(nodes.keys())

    removal_order = []

    # Process deepest nodes first; use node ID as stable tie-breaker
    sorted_nodes = sorted(
        nodes.keys(),
        key=lambda nid: (-nodes[nid]["depth"], nid)
    )

    for nid in sorted_nodes:
        n = nodes[nid]

        # Hard rule: only removable nodes are candidates
        if not n["removable"]:
            continue

        # Hard rule: never remove a node whose symbol is required
        if n["symbol"] in required_symbols_set:
            continue

        # Hard rule: never remove a node that still has retained children
        if any(c in retained for c in children[nid]):
            continue

        # Hard rule: all required symbols must remain represented after removal
        tentative = retained - {nid}
        remaining_symbols = {nodes[rid]["symbol"] for rid in tentative}
        if not required_symbols_set.issubset(remaining_symbols):
            continue

        # Safe to remove
        retained = tentative
        removal_order.append(nid)

    return {
        "retained_ids": sorted(retained),
        "removal_order": removal_order,
    }
```

The algorithm:

1. Validates all required fields and structural integrity (unique IDs, valid parent refs, single root, required symbols present).
2. Sorts candidates deepest-first, node ID as tie-breaker — this ensures leaves are evaluated before their parents, so a parent's "has retained children" check reflects the outcome of prior removals.
3. For each candidate, applies four hard guards in order: must be `removable`, symbol must not be required, must have no retained children, and removing it must leave every required symbol still covered.
4. Returns `retained_ids` (lexicographically sorted) and `removal_order` (the actual sequence of removals).
